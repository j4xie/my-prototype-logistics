## 1. 目标与边界

### 1.1 目标

用户问任何业务问题, 系统**自动**:

1. 拆解问题 (LLM-augmented planner)
2. 决定从哪些 fact/agg 表取数据
3. 通过 entity_id auto-join 多表
4. 调用 C 的 ProvenanceValue 算综合 confidence
5. 返回结果 + Trust Indicator

**前后对比**:

```
v1 (现状): 模板编写者写 SQL
@template
def dish_sales_top_n(schema):
    return f"SELECT product, SUM(amount) FROM agg_product 
            WHERE factory_id='{factory_id}' GROUP BY product ORDER BY 2 DESC LIMIT 20"

v2 (D 后): 模板编写者声明意图, 引擎生成 SQL
@template
class DishSalesTopN(Template):
    requires = ["product_name", "revenue"]
    grain = "product"
    aggregate = "sum(revenue)"
    top_n = 20
    # 引擎自动选 agg_product_period (有此 field), 写 SQL, 算 confidence
```

### 1.2 在 Data Fabric 体系中的角色


### 1.3 in-scope vs out-of-scope

**In Scope**:
- ✅ Query Planner (LLM-augmented, 默认 qwen-plus, qwen-max 仅复杂 case)
- ✅ Logical Query Plan 数据结构 + 校验
- ✅ Logical → Physical SQL Generator (含 auto-join, time-axis 对齐, agg 表优先回退 fact)
- ✅ Confidence Cascade (从 C 的 ProvenanceValue 综合)

- ❌ Cross-tenant query (后期)
- ❌ Streaming / realtime (现状 batch 足够)
- ❌ Query optimizer (PG planner 自带, D 不重做)
- ❌ 全部 44 模板迁移 (高频 20 个先, 低频后期)

```python
# backend/python/smartbi/federated/types.py
from dataclasses import dataclass
from typing import List, Optional, Literal

@dataclass(frozen=True)
class QueryRequest:
    """用户输入."""
    factory_id: str
    question: str                         # 自然语言问题
    date_range: Optional[tuple[date, date]]
    template_hint: Optional[str] = None   # 客户选了某模板时跳过 LLM

@dataclass(frozen=True)
class LogicalQueryPlan:
    """planner 输出: 逻辑查询计划 (待编译为 SQL)."""
    measures: List["MeasureSpec"]
    dimensions: List["DimSpec"]
    filters: List["FilterSpec"]
    # v1.1 修 C-1: 加 date_range (之前 §3.4 引用但未声明)
    # None = 无时间过滤; "row" grain (无聚合, fact 表明细) 也可
    date_range: Optional[tuple[date, date]] = None
    time_grain: Optional[Literal["row", "day", "week", "month", "quarter", "year"]] = None
    sort: Optional["SortSpec"] = None
    limit: Optional[int] = None
    confidence_required: float = 0.0
    
    # v1.1 修 C-3: 反序列化辅助
    def to_dict(self) -> dict:
        from dataclasses import asdict
        d = asdict(self)
        # tuple → list (JSON 兼容); date → ISO string
        if self.date_range:
            d["date_range"] = [self.date_range[0].isoformat(), self.date_range[1].isoformat()]
        return d
    
    @classmethod
    def from_dict(cls, d: dict) -> "LogicalQueryPlan":
        from datetime import date
        return cls(
            measures=[MeasureSpec(**m) for m in d.get("measures", [])],
            dimensions=[DimSpec(**dim) for dim in d.get("dimensions", [])],
            filters=[FilterSpec(**f) for f in d.get("filters", [])],
            date_range=tuple(date.fromisoformat(s) for s in d["date_range"]) if d.get("date_range") else None,
            time_grain=d.get("time_grain"),
            sort=SortSpec(**d["sort"]) if d.get("sort") else None,
            limit=d.get("limit"),
            confidence_required=d.get("confidence_required", 0.0),
        )
    
@dataclass(frozen=True)
class MeasureSpec:
    canonical_field: str                  # 'revenue' / 'qty_sold' / 'rating'
    aggregation: Literal["sum", "avg", "count", "max", "min", "count_distinct"]
    alias: str                            # 输出列名
    
@dataclass(frozen=True)
class DimSpec:
    canonical_field: str                  # 'product_name' / 'store_name'
    entity_type: Optional[str] = None     # auto-join 时用 (product / store / staff)
    
@dataclass(frozen=True)
class FilterSpec:
    canonical_field: str
    op: Literal["=", "!=", ">", "<", ">=", "<=", "IN", "BETWEEN", "LIKE"]
    value: Any
    
@dataclass(frozen=True)
class QueryResult:
    """最终输出."""
    columns: List[str]
    rows: List[dict]
    confidence: float                     # 综合 confidence (来自 C provenance)
    sources: List[str]                    # 数据来源 (用于 Trust UI)
    sql_executed: str                     # 透明可审计
    elapsed_ms: int
    plan: LogicalQueryPlan                # 给前端展示用
```

### 2.3 LLM Planner Prompt 设计

```python
# backend/python/smartbi/federated/llm_planner.py
PLANNER_SYSTEM_PROMPT = """
你是数据查询计划专家. 把用户问题翻译成结构化查询计划.

可用字段 (factory 上传过的): {available_fields}
可用模板 (高频): {template_registry}
当前日期: {today}

输出格式 (JSON):
{
  "measures": [{"canonical_field": "...", "aggregation": "sum|avg|count|...", "alias": "..."}],
  "dimensions": [{"canonical_field": "...", "entity_type": "product|store|staff"}],
  "filters": [{"canonical_field": "...", "op": "=|>|...", "value": "..."}],
  "time_grain": "day|week|month|year|null",
  "sort": {"by": "...", "direction": "asc|desc"},
  "limit": <int 或 null>
}

注意:
- canonical_field 必须在 available_fields 里, 否则 plan 失败
- entity_type 用于 auto-join, 加上才能跨表
- time_grain 仅当 dimensions 含 'date' 时有效
- 不能用的字段或操作请明确返 {"error": "..."}
"""

_planner_inflight: dict[str, asyncio.Future] = {}  # v1.1 修 C-4

def _normalize_question(q: str) -> str:
    """v1.1: 问题归一化, 防同意 不同 hash. 全角→半角, 大小写, 空白."""
    import unicodedata
    q = unicodedata.normalize("NFKC", q)  # 全角→半角
    q = q.lower().strip()
    q = re.sub(r'\s+', ' ', q)  # 折叠多空白
    return q

def _build_cache_key(request: QueryRequest, capability: Set[str]) -> str:
    """v1.1 修 C-5: cache key 含 capability hash, capability 变 invalidate."""
    norm_q = _normalize_question(request.question)
    cap_hash = sha256(",".join(sorted(capability)).encode()).hexdigest()[:8]
    date_str = ""
    if request.date_range:
        date_str = f"{request.date_range[0].isoformat()}|{request.date_range[1].isoformat()}"
    raw = f"{request.factory_id}|{norm_q}|{date_str}|{cap_hash}"
    return sha256(raw.encode()).hexdigest()

async def llm_plan(request: QueryRequest, capability: Set[str], pool) -> LogicalQueryPlan:
    """调 LLM 生成 plan. 复用 Week 5 narrative_cache 缓存.
    
    v1.1 修复:
    - C-3: 用 LogicalQueryPlan.from_dict 反序列化 (nested dataclass)
    - C-4: inflight dedup 防并发同问双调 LLM
    - C-5: cache_key 含 capability_hash, A 端 capability 变更自动 invalidate
    """
    cache_key = _build_cache_key(request, capability)
    
    # 1. 缓存命中
    cached_plan = await NarrativeCache(pool).get(cache_key)
    if cached_plan:
        return LogicalQueryPlan.from_dict(cached_plan)  # v1.1 修 C-3
    
    # 2. v1.1 修 C-4: inflight dedup
    if cache_key in _planner_inflight:
        return await _planner_inflight[cache_key]
    
    # 3. 真正调 LLM (v1.1: 包裹在 inflight Future 内防并发)
    future: asyncio.Future = asyncio.Future()
    _planner_inflight[cache_key] = future
    try:
        plan = await _llm_plan_inner(request, capability, pool, cache_key)
        future.set_result(plan)
        return plan
    except Exception as e:
        future.set_exception(e)
        raise
    finally:
        _planner_inflight.pop(cache_key, None)

async def _llm_plan_inner(request, capability, pool, cache_key) -> LogicalQueryPlan:
    settings = get_settings()
    model = _select_planner_model(request.question, capability)
    template_registry = await _load_template_registry(pool)
    
    response = await llm_client.chat(
        system=PLANNER_SYSTEM_PROMPT.format(
            available_fields=sorted(capability),
            template_registry=template_registry,
            today=date.today(),
        ),
        user=request.question,
        model=model,
        max_tokens=800,
    )
    
    plan_json = json.loads(response)
    if "error" in plan_json:
        raise PlannerError(plan_json["error"])
    
    plan = LogicalQueryPlan.from_dict(plan_json)  # v1.1 修 C-3
    _validate_plan(plan, capability)
    
    # cache 24h (v1.1 修 C-3: to_dict 处理 nested + tuple/date)
    await NarrativeCache(pool).set(cache_key, plan.to_dict(), ttl_hours=24)
    
    await budget_tracker.consume_with_feature(
        request.factory_id, response.tokens, "query_planner"
    )
    
    return plan
```

### 2.4 Plan validation

```python
def _validate_plan(plan: LogicalQueryPlan, capability: Set[str]):
    """plan 字段必须在 capability 里 (来自 A v1.4 contract)."""
    used_fields = set()
    for m in plan.measures:
        used_fields.add(m.canonical_field)
    for d in plan.dimensions:
        used_fields.add(d.canonical_field)
    for f in plan.filters:
        used_fields.add(f.canonical_field)
    
    missing = used_fields - capability
    if missing:
        raise PlannerError(
            f"Plan 引用 capability 未提供的字段: {missing}. "
            f"客户上传含这些字段的数据可解锁此查询."
        )
    
    # entity_type 必须在已知 dim 表
    KNOWN_ENTITIES = {"product", "store", "staff", "ingredient", "subject"}
    for d in plan.dimensions:
        if d.entity_type and d.entity_type not in KNOWN_ENTITIES:
            raise PlannerError(f"未知 entity_type: {d.entity_type}")
```

### 2.5 qwen-plus 默认, qwen-max 仅复杂 case

```python
def _select_planner_model(question: str, capability: Set[str]) -> str:
    """复杂查询用 qwen-max, 否则 qwen-plus.
    
    复杂判定:
    - 字段引用 ≥ 5 个
    - 含 cross-fact join 词 (含成本/毛利/评分 综合的)
    - 含时间序列 + 同环比 双轴
    """
    complex_keywords = ["毛利", "评分", "同比", "环比", "对比", "占比"]
    if any(kw in question for kw in complex_keywords):
        return "qwen-max"
    if len(question) > 100:  # 长问题倾向复杂
        return "qwen-max"
    return "qwen-plus"
```

## 3. Logical → Physical SQL Generator

### 3.1 表选择策略 (粗 agg 优先, 缺时回退 fact)

```python
# backend/python/smartbi/federated/sql_generator.py
TABLE_PRIORITY = {
    # field → [(table, grain, where_clause)]  按粒度从粗到细排序
    "revenue": [
        ("agg_product_period", "month", ""),       # 最粗 (product × period)
        ("agg_daily", "day", ""),                  # 中等 (date)
        ("fact_pos_item", "row", "WHERE qty > 0"), # 最细 (per item)
    ],
    "qty_sold": [
        ("agg_product_period", "month", ""),
        ("fact_pos_item", "row", ""),
    ],
    "rating": [
        ("dim_review_summary", "month", ""),
        ("fact_review_event", "row", ""),
    ],
    "stock_qty": [
        ("fact_inventory_snapshot", "day", ""),
    ],
    "voucher_amount": [
        ("fact_finance_voucher", "day", ""),
    ],
    # v1.1 修 S-3: 补全 B v1.2 引入的字段
    "review_text": [
        ("fact_review_event", "row", ""),
    ],
    "review_count": [
        ("dim_review_summary", "month", ""),
    ],
    "ingredient_qty": [
        ("fact_inventory_snapshot", "day", ""),
    ],
    "voucher_subject": [
        ("fact_finance_voucher", "day", ""),
    ],
    # 客单价 / 账单数 — 来自现有 fact_pos_transaction
    "bill_count": [
        ("agg_daily", "day", ""),
        ("fact_pos_transaction", "row", ""),
    ],
    "avg_bill_value": [
        ("agg_daily", "day", ""),
        ("fact_pos_transaction", "row", ""),
    ],
}

def choose_source_table(plan: LogicalQueryPlan) -> str:
    """根据 measures + grain 选最优表.
    
    v1.1 修 C-6: time_grain=None / "row" 时不再默认 month, 直接走最细 fact 表.
    """
    primary_measure = plan.measures[0]
    candidates = TABLE_PRIORITY.get(primary_measure.canonical_field, [])
    if not candidates:
        raise PlannerError(f"无 source table for {primary_measure.canonical_field}")
    
    requested_grain = plan.time_grain
    GRAIN_ORDER = {"row": 0, "day": 1, "week": 2, "month": 3, "quarter": 4, "year": 5}
    
    # v1.1 修 C-6:
    # - time_grain == "row" → 用最细 fact 表 (用户要 raw 行数据)
    # - time_grain is None + 没有 date dimension → 也用最细 (无时间分组)
    # - time_grain is None + 有 date dimension → fallback 'day' (合理默认)
    if requested_grain == "row":
        return candidates[-1][0]
    if requested_grain is None:
        has_date_dim = any(d.canonical_field == "date" for d in plan.dimensions)
        requested_grain = "day" if has_date_dim else "row"
        if requested_grain == "row":
            return candidates[-1][0]
    
    for table, table_grain, where in candidates:
        if GRAIN_ORDER[table_grain] <= GRAIN_ORDER[requested_grain]:
            return table
    
    return candidates[-1][0]
```

### 3.2 Auto-join via entity_id (v1.1 修 C-1: alias 防冲突)

```python
# v1.1: 显式 alias 映射, 避免 staff/store 都首字母 's' 冲突
ENTITY_ALIAS_MAP = {
    "product": "prd",
    "store": "sto",
    "staff": "stf",
    "ingredient": "ing",
    "subject": "sub",
    "customer": "cus",
    "supplier": "sup",
}

def auto_join(primary_table: str, dimensions: List[DimSpec]) -> tuple[str, dict[str, str]]:
    """v1.1 修 C-1: 显式 3 字母 alias 防冲突, 返 (sql_fragment, alias_map)."""
    joins_parts = []
    aliases = {}
    
    for d in dimensions:
        if not d.entity_type:
            continue
        dim_table = f"dim_{d.entity_type}"
        alias = ENTITY_ALIAS_MAP.get(d.entity_type)
        if not alias:
            raise PlannerError(f"Unknown entity_type: {d.entity_type}")
        aliases[d.entity_type] = alias
        joins_parts.append(
            f" LEFT JOIN {dim_table} {alias} "
            f"ON {alias}.{d.entity_type}_id = primary.{d.entity_type}_id "
            # v1.1 加: 排除 sentinel id=0 (来自 A v1.4 / C v1.2 sentinel scheme, 修 S-2)
            f"AND {alias}.{d.entity_type}_id != 0"
        )
    
    return "".join(joins_parts), aliases
```

### 3.3 Time-axis 自动对齐

```python
def time_grain_clause(grain: str, date_field: str = "date") -> str:
    """生成 GROUP BY 时间表达式."""
    if grain == "day":
        return date_field
    elif grain == "week":
        return f"date_trunc('week', {date_field})"
    elif grain == "month":
        return f"date_trunc('month', {date_field})"
    elif grain == "quarter":
        return f"date_trunc('quarter', {date_field})"
    elif grain == "year":
        return f"date_trunc('year', {date_field})"
    raise ValueError(f"Unknown grain: {grain}")
```

### 3.4 完整 SQL 生成 (v1.1 修 C-1 多 bug + S-1 SQL 注入)

**v1.1 关键修复**:
- **S-1 SQL 注入**: aggregation / sort_by / op 全部白名单, 不在白名单 reject plan
- **C-1**: GROUP BY 用 dimensions 显式枚举 (不再"含 ( 启发式"), `auto_join` 解构正确, alias 用 ENTITY_ALIAS_MAP, filter 列考虑 entity_type 解析

```python
# v1.1 修 S-1: 白名单 (任何不在白名单的输入直接 reject, 阻止 SQL 注入)
ALLOWED_AGGREGATIONS = {"sum", "avg", "count", "max", "min", "count_distinct"}
ALLOWED_SORT_DIRECTIONS = {"asc", "desc"}
ALLOWED_FILTER_OPS = {"=", "!=", ">", "<", ">=", "<=", "IN", "BETWEEN", "LIKE"}
MAX_LIMIT = 10_000

# v1.1: identifier 白名单从 ALIAS_TO_ATTR (A v1.4 §2.1) 全部 canonical 字段 + entity_id 字段
ALLOWED_FIELDS = set(ALIAS_TO_ATTR.values()) | {
    "product_id", "store_id", "staff_id", "ingredient_id", "subject_id", "customer_id"
}

def _validate_identifier(name: str, kind: str) -> str:
    """白名单校验, 不通过 raise. 任何 SQL identifier 都过此函数."""
    if name not in ALLOWED_FIELDS:
        raise PlannerError(f"非法 {kind} 标识符: {name!r} (不在白名单)")
    return name

def generate_sql(plan: LogicalQueryPlan, factory_id: str) -> tuple[str, list]:
    """v1.1 修 C-1 多 bug + S-1 SQL 注入防护."""
    # 1. 选表
    primary_table = choose_source_table(plan)
    
    # 2. JOIN (修 C-1: 解构正确)
    joins_sql, aliases = auto_join(primary_table, plan.dimensions)
    
    # 3. SELECT (v1.1: 显式标记 dim 还是 measure, GROUP BY 用此表)
    select_parts = []
    group_by_parts = []
    
    for d in plan.dimensions:
        col = _validate_identifier(d.canonical_field, "dimension")
        if d.entity_type:
            alias = aliases[d.entity_type]
            expr = f"{alias}.name"
            sql_part = f"{expr} AS {col}"
            select_parts.append(sql_part)
            group_by_parts.append(expr)  # 显式记录, 不靠启发式
        elif col == "date":
            # 时间维度
            if plan.time_grain == "row":
                expr = "primary.date"
                sql_part = "primary.date AS event_date"
            else:
                expr = time_grain_clause(plan.time_grain or "day", "primary.date")
                sql_part = f"{expr} AS time_period"
            select_parts.append(sql_part)
            group_by_parts.append(expr)
        else:
            expr = f"primary.{col}"
            select_parts.append(expr)
            group_by_parts.append(expr)
    
    for m in plan.measures:
        agg = m.aggregation.lower()
        if agg not in ALLOWED_AGGREGATIONS:
            raise PlannerError(f"非法 aggregation: {agg!r}")
        col = _validate_identifier(m.canonical_field, "measure")
        alias_col = _validate_identifier(m.alias, "measure_alias")
        if agg == "count_distinct":
            select_parts.append(f"COUNT(DISTINCT primary.{col}) AS {alias_col}")
        else:
            select_parts.append(f"{agg.upper()}(primary.{col}) AS {alias_col}")
    
    # 4. WHERE (v1.1: filter 列考虑 entity_type 把列指到正确的 alias)
    params = []
    where_parts = ["primary.factory_id = current_setting('app.factory_id')"]
    
    for f in plan.filters:
        if f.op not in ALLOWED_FILTER_OPS:
            raise PlannerError(f"非法 filter op: {f.op!r}")
        col = _validate_identifier(f.canonical_field, "filter")
        # filter 列在 primary 还是 dim_*? 看是不是 entity_type 的字段 (e.g. store_name 在 dim_store)
        # v1.1: 简化 — filter 全在 primary, dim 字段做 join 后比较 (用别名)
        if col in {"store_name", "product_name", "staff_name"}:
            # 这些是 dim_*.name, 必须 JOIN 后才能 filter
            entity = col.replace("_name", "")
            alias = aliases.get(entity)
            if not alias:
                raise PlannerError(f"filter 列 {col} 需 JOIN dim_{entity} 但 plan 没 dimensions 引用")
            params.append(f.value)
            where_parts.append(f"{alias}.name {f.op} ${len(params)}")
        else:
            params.append(f.value)
            where_parts.append(f"primary.{col} {f.op} ${len(params)}")
    
    # date_range (v1.1: plan.date_range 现已在 dataclass)
    if plan.date_range:
        params.extend(plan.date_range)
        where_parts.append(f"primary.date BETWEEN ${len(params)-1} AND ${len(params)}")
    
    # 5. GROUP BY (v1.1: 用显式 group_by_parts, 不靠启发式)
    has_aggregation = any(
        m.aggregation.upper() != "ROW_OUTPUT" for m in plan.measures
    )
    group_by_clause = ""
    if group_by_parts and has_aggregation and plan.time_grain != "row":
        group_by_clause = " GROUP BY " + ", ".join(group_by_parts)
    
    # 6. ORDER BY (S-1: sort.by 白名单)
    order_clause = ""
    if plan.sort:
        sort_col = _validate_identifier(plan.sort.by, "sort_by")
        if plan.sort.direction.lower() not in ALLOWED_SORT_DIRECTIONS:
            raise PlannerError(f"非法 sort direction: {plan.sort.direction!r}")
        order_clause = f" ORDER BY {sort_col} {plan.sort.direction.upper()}"
    
    # 7. LIMIT (S-1: int 强制 + max cap)
    limit_clause = ""
    if plan.limit is not None:
        if not isinstance(plan.limit, int) or plan.limit <= 0 or plan.limit > MAX_LIMIT:
            raise PlannerError(f"非法 limit: {plan.limit}")
        limit_clause = f" LIMIT {plan.limit}"
    
    sql = f"""
        SELECT {', '.join(select_parts)}
          FROM {primary_table} primary
          {joins_sql}
         WHERE {' AND '.join(where_parts)}
         {group_by_clause}
         {order_clause}
         {limit_clause}
    """.strip()
    return sql, params
```

## 4. Confidence Cascade

### 4.1 多源 confidence 综合 (v1.1 修 C-2 RLS 一致)

```python
async def compute_query_confidence(
    plan: LogicalQueryPlan, factory_id: str, conn
) -> tuple[float, List[str]]:
    """Query 的 confidence = 所有依赖字段 confidence 的加权综合.
    
    v1.1 修复:
    - C-2 RLS 一致: 不用 explicit WHERE factory_id, 走 set_config 让 RLS 过滤
    - 主 measure 权重高于 dim (避免 outlier dim 字段拖整查询 confidence)
    - sample 大小 50 (10 太小, 上传分布偏)
    - fact 表数据 confidence 改 0.90 不是 0.95 (诚实, 留余地)
    """
    # v1.1 C-2: 调用方负责 set_config; 这里假设已设
    
    field_confidences = []  # [(confidence, weight)]
    sources = set()
    
    # 主 measure 权重 1.0, dim 权重 0.5
    for m in plan.measures:
        confs = await _sample_confidence(conn, factory_id, m.canonical_field, sample_size=50)
        if confs:
            avg = sum(c[0] for c in confs) / len(confs)
            field_confidences.append((avg, 1.0))
            sources.update(c[1] for c in confs)
        else:
            field_confidences.append((0.90, 1.0))  # fact 默认 0.90 不是 0.95
    
    for d in plan.dimensions:
        if d.entity_type:
            continue  # entity 名字来自 dim_*, 假定 1.0 (已实体解析)
        confs = await _sample_confidence(conn, factory_id, d.canonical_field, sample_size=20)
        if confs:
            avg = sum(c[0] for c in confs) / len(confs)
            field_confidences.append((avg, 0.5))
            sources.update(c[1] for c in confs)
    
    if not field_confidences:
        return 0.90, []
    
    # 加权平均不再用 min (修 D5 risk + S-4)
    total_weight = sum(w for _, w in field_confidences)
    overall = sum(c * w for c, w in field_confidences) / total_weight
    return round(overall, 3), sorted(sources)

async def _sample_confidence(conn, factory_id: str, field_name: str, sample_size: int = 50):
    """RLS 已 set, 不用 WHERE factory_id."""
    return await conn.fetch("""
        SELECT confidence, source_type FROM field_provenance
         WHERE field_name = $1
           AND superseded_by_id IS NULL
         ORDER BY created_at DESC LIMIT $2
    """, field_name, sample_size)
```

### 4.2 输出到 Trust UI

```python
async def execute_query(
    plan: LogicalQueryPlan, factory_id: str, pool, timeout_ms: int = 30000
) -> QueryResult:
    """v1.1 修复:
    - C-2 RLS 一致: confidence cascade 也在同事务内, 共享 set_config
    - S-7 timeout: SET LOCAL statement_timeout 防 plan poisoning
    """
    sql, params = generate_sql(plan, factory_id)
    
    async with pool.acquire() as conn:
        async with conn.transaction():
            await conn.execute("SELECT set_config('app.factory_id', $1, true)", factory_id)
            await conn.execute(f"SET LOCAL statement_timeout = {int(timeout_ms)}")
            
            t0 = time.time()
            try:
                rows = await conn.fetch(sql, *params)
            except asyncpg.exceptions.QueryCanceledError:
                raise PlannerError(f"查询超过 {timeout_ms}ms 超时, 计划已从 cache 排除")
            elapsed_ms = int((time.time() - t0) * 1000)
            
            # v1.1 C-2: 同事务跑 confidence, RLS 一致 (compute_query_confidence 不再传 factory_id 参数)
            confidence, sources = await compute_query_confidence(plan, factory_id, conn)
    
    return QueryResult(
        columns=[c.canonical_field if hasattr(c, 'canonical_field') else c.alias 
                 for c in plan.dimensions + plan.measures],
        rows=[dict(r) for r in rows],
        confidence=confidence,
        sources=sources,
        sql_executed=sql,
        elapsed_ms=elapsed_ms,
        plan=plan,
    )
```
